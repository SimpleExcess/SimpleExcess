def create_styled_workbook(df: pd.DataFrame) -> bytes:
    """Create a nicely formatted Excel workbook from DataFrame."""
    wb = Workbook()
    ws = wb.active
    ws.title = "Excess Inventory"

    # Styles
    header_font = Font(bold=True, color="FFFFFF", size=11)
    header_fill = PatternFill(start_color="1F4E79", end_color="1F4E79", fill_type="solid")
    thin_border = Border(
        left=Side(style="thin"),
        right=Side(style="thin"),
        top=Side(style="thin"),
        bottom=Side(style="thin")
    )
    center_align = Alignment(horizontal="center", vertical="center", wrap_text=True)

    # Header
    for col_idx, col_name in enumerate(REQUIRED_COLUMNS, 1):
        cell = ws.cell(row=1, column=col_idx, value=col_name)
        cell.font = header_font
        cell.fill = header_fill
        cell.alignment = center_align
        cell.border = thin_border

    # Data rows
    for row_idx, row in enumerate(df.itertuples(index=False), 2):
        for col_idx, value in enumerate(row, 1):
            cell = ws.cell(row=row_idx, column=col_idx, value=value if pd.notna(value) else "")
            cell.border = thin_border
            cell.alignment = Alignment(vertical="center", wrap_text=True)

    # Column widths
    widths = [22, 12, 20, 22, 14, 16]
    for i, w in enumerate(widths, 1):
        ws.column_dimensions[get_column_letter(i)].width = w

    # Freeze header
    ws.freeze_panes = "A2"

    buffer = io.BytesIO()
    wb.save(buffer)
    buffer.seek(0)
    return buffer.getvalue()


def normalize_dataframe(df: pd.DataFrame) -> pd.DataFrame:
    """Map common column name variations to standard names."""
    col_map = {}
    lower_cols = {c.lower().strip(): c for c in df.columns}

    mapping_rules = {
        "Part Number": ["part number", "part no", "part#", "mpn", "manufacturer part number", "pn", "part_number"],
        "Quantity": ["quantity", "qty", "qty.", "amount", "stock", "available"],
        "Manufacturer": ["manufacturer", "mfr", "mfg", "brand", "maker", "vendor"],
        "Internal Part Number": ["internal part number", "internal pn", "ipn", "customer part number", "cpn", "internal", "sku"],
        "Date Code": ["date code", "datecode", "dc", "date", "lot date", "week code"],
        "Packaging": ["packaging", "package", "pkg", "packing", "form", "reel", "tray"]
    }

    for std_name, variants in mapping_rules.items():
        for v in variants:
            if v in lower_cols:
                col_map[lower_cols[v]] = std_name
                break

    df = df.rename(columns=col_map)

    # Ensure all required columns exist
    for col in REQUIRED_COLUMNS:
        if col not in df.columns:
            df[col] = ""

    return df[REQUIRED_COLUMNS].copy()


def extract_from_ocr_text(text: str) -> dict:
    """
    Heuristic extraction of part data from OCR text of an electronics label.
    Returns a dict with keys matching REQUIRED_COLUMNS.
    """
    result = {col: "" for col in REQUIRED_COLUMNS}
    if not text or not text.strip():
        return result

    lines = [l.strip() for l in text.splitlines() if l.strip()]
    full = " ".join(lines)

    # Quantity patterns: QTY: 1000, Qty 500, Quantity: 2500, 1000 pcs, etc.
    qty_patterns = [
        r"(?:qty|quantity|q\.?ty\.?)\s*[:=]?\s*(\d{1,7})",
        r"(\d{2,7})\s*(?:pcs|pieces|ea|each|units?)",
        r"\b(\d{3,6})\b(?=\s*(?:pcs|pieces)?\s*$)"
    ]
    for pat in qty_patterns:
        m = re.search(pat, full, re.IGNORECASE)
        if m:
            result["Quantity"] = m.group(1)
            break

    # Date code: common formats YYWW, YYYYWW, or "Date Code: 2312", "DC: 2245"
    dc_patterns = [
        r"(?:date\s*code|dc|d/?c)\s*[:=]?\s*([0-9]{4,6})",
        r"\b(20[1-2][0-9][0-5][0-9])\b",  # 2010-2029 weeks
        r"\b([0-2][0-9][0-5][0-9])\b"       # YYWW
    ]
    for pat in dc_patterns:
        m = re.search(pat, full, re.IGNORECASE)
        if m:
            result["Date Code"] = m.group(1)
            break

    # Packaging keywords
    pkg_keywords = ["reel", "tape & reel", "tray", "tube", "bulk", "cut tape", "ammo pack", "bag"]
    for kw in pkg_keywords:
        if re.search(rf"\b{re.escape(kw)}\b", full, re.IGNORECASE):
            result["Packaging"] = kw.title()
            break

    # Manufacturer: look for common known manufacturers or first prominent line
    known_mfrs = [
        "Texas Instruments", "TI", "Analog Devices", "ADI", "Maxim", "Microchip",
        "STMicroelectronics", "ST", "Infineon", "NXP", "ON Semiconductor", "onsemi",
        "Vishay", "Murata", "TDK", "Samsung", "SK Hynix", "Micron", "Intel",
        "Broadcom", "Qualcomm", "Xilinx", "AMD", "Lattice", "Cypress", "Renesas",
        "Rohm", "Diodes Inc", "Bourns", "Yageo", "Kemet", "AVX", "TE Connectivity"
    ]
    for mfr in known_mfrs:
        if re.search(rf"\b{re.escape(mfr)}\b", full, re.IGNORECASE):
            result["Manufacturer"] = mfr
            break

    # Part Number: often the longest alphanumeric string that looks like a PN
    # Common patterns: letters+numbers, with possible - / _
    pn_candidates = re.findall(r"\b([A-Z0-9][A-Z0-9\-_/]{4,30})\b", full)
    # Prefer ones that contain both letters and digits
    scored = []
    for cand in pn_candidates:
        if re.search(r"[A-Za-z]", cand) and re.search(r"[0-9]", cand):
            score = len(cand)
            # boost if looks like typical MPN
            if re.search(r"[A-Z]{2,}", cand):
                score += 5
            scored.append((score, cand))
    if scored:
        scored.sort(reverse=True)
        result["Part Number"] = scored[0][1]

    # Internal / Customer PN sometimes labeled
    ipn_match = re.search(r"(?:internal|customer|cpn|ipn|sku)\s*(?:part\s*)?(?:no|number|#)?\s*[:=]?\s*([A-Z0-9\-_/]{3,25})", full, re.IGNORECASE)
    if ipn_match:
        result["Internal Part Number"] = ipn_match.group(1)

    return result


def send_email_with_attachment(
    to_addr: str,
    subject: str,
    body: str,
    attachment_bytes: bytes,
    attachment_filename: str,
    smtp_server: str,
    smtp_port: int,
    smtp_user: str,
    smtp_password: str
) -> tuple[bool, str]:
    """Send email with Excel attachment. Returns (success, message)."""
    try:
        msg = MIMEMultipart()
        msg["From"] = smtp_user
        msg["To"] = to_addr
        msg["Subject"] = subject

        msg.attach(MIMEText(body, "plain"))

        part = MIMEBase("application", "vnd.openxmlformats-officedocument.spreadsheetml.sheet")
        part.set_payload(attachment_bytes)
        encoders.encode_base64(part)
        part.add_header("Content-Disposition", f"attachment; filename={attachment_filename}")
        msg.attach(part)

        with smtplib.SMTP(smtp_server, smtp_port, timeout=30) as server:
            server.starttls()
            server.login(smtp_user, smtp_password)
            server.send_message(msg)

        return True, f"Successfully sent to {to_addr}"
    except Exception as e:
        return False, str(e)

# ===== TAB 2: Label Images =====
with tab2:
    st.header("Upload Label Photos")
    st.markdown(
        "Upload clear photos of component labels / packaging. The app will run OCR "
        "and attempt to extract the fields. You can then correct any values before adding."
    )

    uploaded_images = st.file_uploader(
        "Choose image(s) of labels",
        type=["png", "jpg", "jpeg", "bmp", "tiff", "webp"],
        accept_multiple_files=True,
        key="image_uploader"
    )

    if uploaded_images:
        for idx, img_file in enumerate(uploaded_images):
            st.subheader(f"Image {idx + 1}: {img_file.name}")
            col_img, col_data = st.columns([1, 1.2])

            try:
                image = Image.open(img_file)
                if image.mode not in ("RGB", "L"):
                    image = image.convert("RGB")
            except Exception as e:
                st.error(f"Could not open image: {e}")
                continue

            with col_img:
                st.image(image, caption=img_file.name, use_container_width=True)

            with col_data:
                ocr_text = ""
                extracted = {col: "" for col in REQUIRED_COLUMNS}
                with st.spinner("Running OCR..."):
                    try:
                        ocr_image = image.convert("RGB") if image.mode != "RGB" else image.copy()
                        w, h = ocr_image.size
                        if max(w, h) < 800:
                            scale = 800 / max(w, h)
                            try:
                                resample = Image.Resampling.LANCZOS
                            except AttributeError:
                                resample = Image.LANCZOS
                            ocr_image = ocr_image.resize(
                                (int(w * scale), int(h * scale)), resample
                            )
                        ocr_text = pytesseract.image_to_string(ocr_image)
                        extracted = extract_from_ocr_text(ocr_text)
                    except Exception as e:
                        st.warning(f"OCR failed: {e}. You can still enter values manually.")

                with st.expander("Raw OCR text", expanded=False):
                    st.text(ocr_text.strip() if ocr_text and ocr_text.strip() else "(no text detected)")

                st.markdown("**Extracted / Editable fields**")
                # Sanitize key (filenames can contain chars Streamlit rejects)
                safe_name = re.sub(r"[^a-zA-Z0-9_-]", "_", img_file.name)[:40]
                key_prefix = f"img_{idx}_{safe_name}"

                pn = st.text_input("Part Number", value=extracted.get("Part Number", ""), key=f"{key_prefix}_pn")
                qty = st.text_input("Quantity", value=extracted.get("Quantity", ""), key=f"{key_prefix}_qty")
                mfr = st.text_input("Manufacturer", value=extracted.get("Manufacturer", ""), key=f"{key_prefix}_mfr")
                ipn = st.text_input("Internal Part Number", value=extracted.get("Internal Part Number", ""), key=f"{key_prefix}_ipn")
                dc = st.text_input("Date Code", value=extracted.get("Date Code", ""), key=f"{key_prefix}_dc")
                pkg = st.text_input("Packaging", value=extracted.get("Packaging", ""), key=f"{key_prefix}_pkg")

                if st.button("➕ Add this label to inventory", key=f"{key_prefix}_add"):
                    new_row = {
                        "Part Number": (pn or "").strip(),
                        "Quantity": (qty or "").strip(),
                        "Manufacturer": (mfr or "").strip(),
                        "Internal Part Number": (ipn or "").strip(),
                        "Date Code": (dc or "").strip(),
                        "Packaging": (pkg or "").strip(),
                    }
                    new_df = pd.DataFrame([new_row], columns=REQUIRED_COLUMNS)
                    base = st.session_state.inventory_df
                    if base is None or base.empty:
                        st.session_state.inventory_df = new_df
                    else:
                        st.session_state.inventory_df = pd.concat(
                            [base, new_df], ignore_index=True
                        )
                    st.success("Row added!")
                    st.rerun()

            st.divider()
